echo $VIRTUAL_ENV       --it checks that virtual environment exist or not 
which python3

python -m venv --help  # just to confirm venv module exists

wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh

export PATH="/home/iid/miniconda3/bin:$PATH"
eval "$(/home/iid/miniconda3/bin/conda shell.bash hook)"


cd projectname

conda tos accept --override-channels --channel https://repo.anaconda.com/pkgs/main
conda tos accept --override-channels --channel https://repo.anaconda.com/pkgs/r

conda create -y -n "agentic311" python=3.13.5

conda activate agentic311
conda deactivate                  # if you want to deactivate

pip install jupyter notebook ipykernel langchain langgraph openai tiktoken httpx python-dotenv
python -m ipykernel install --user --name=agentic311 --display-name "Python 3.13.5 (agentic311)"

mkdir -p src notebooks .vscode
cat > .vscode/settings.json <<EOF
{
  "python.defaultInterpreterPath": "$HOME/miniconda3/envs/$ENV_NAME/bin/python",
  "jupyter.kernels.filter": ["Python 3.11 ($ENV_NAME)"]
}
EOF

jupyter kernelspec list


jupyter --version
